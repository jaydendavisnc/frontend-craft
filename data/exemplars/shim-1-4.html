Summary.tsx
// @figma Checkout / Order summary
export function Summary({ order }: Props) {
  return (
    <Card>
      <Title>Order summary</Title>
      {order.items.map(i => (
        <Row key={i.sku} item={i} />
      ))}
      <PromoField code={order.promo} />
      <Total amount={order.total} />
    </Card>
  )
}
